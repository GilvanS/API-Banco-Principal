Detalhes Técnicos dos Novos Endpoints (Mock)
1) POST /api/v1/transactions/transfer-bank
Request: { agencia: string, conta: string, valor: number, descricao?: string }
Response: { success: true, message: "Transferência bancária realizada", data: { saldoAtual, transacaoId } }

2) POST /api/v1/pix/keys
Request: { type: "CPF"|"EMAIL", key: string }
Response: { success: true, message: "Chave PIX cadastrada", data: { id, type, key } }

3) GET /api/v1/pix/keys
Response: { success: true, data: [ { id, type, key } ] }

4) POST /api/v1/transactions/transfer-pix
Request: { key: string, valor: number, descricao?: string }
Response: { success: true, message: "Transferência PIX realizada", data: { saldoAtual, transacaoId } }

5) GET /api/v1/account/invoice
Response: { success: true, data: { total: number, vencimento: string, itens: [ { id, descricao, valor, data } ] } }

6) POST /api/v1/transactions/pay-bill
Request: { valor: number }
Response: { success: true, message: "Fatura paga com sucesso", data: { creditoUtilizado, saldoAtual } }

7) POST /api/v1/payments/boleto
Request: { linhaDigitavel: string, valor: number }
Response: { success: true, message: "Boleto pago com sucesso", data: { comprovanteId } }

8) POST /api/v1/cards/block
Request: { blocked: boolean }
Response: { success: true, message: "Cartão atualizado", data: { blocked } }

9) POST /api/v1/cards/reissue
Request: {}
Response: { success: true, message: "Solicitação de 2ª via registrada", data: { protocolo } }

10) GET /api/v1/investments
Response: { success: true, data: [ { id, nome, valorInvestido, valorAtual } ] }

11) POST /api/v1/investments/redeem
Request: { investmentId: string, valor: number }
Response: { success: true, message: "Resgate efetuado", data: { saldoAtual, investimento: { id, valorAtual } } }
